function configurarOrion() {
  const spreadsheet = SpreadsheetApp.getActiveSpreadsheet();
  if (!spreadsheet) throw new Error('Abra este Apps Script pela planilha Orion Finance 2.0.');
  const props = PropertiesService.getScriptProperties();
  props.setProperty(ORION.propertySpreadsheetId, spreadsheet.getId());
  let token = props.getProperty(ORION.propertyApiToken);
  if (!token) {
    token = Utilities.getUuid() + Utilities.getUuid().replace(/-/g,'');
    props.setProperty(ORION.propertyApiToken, token);
  }
  atualizarEstruturaOrionV18();
  Logger.log('ORION CONFIGURADO COM SUCESSO');
  Logger.log('Token da API: ' + token);
}

function mostrarTokenOrion() {
  const token = PropertiesService.getScriptProperties().getProperty(ORION.propertyApiToken);
  if (!token) throw new Error('Execute configurarOrion().');
  Logger.log('Token da API: ' + token);
}

function atualizarEstruturaOrionV18() {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  let backup = null;
  try {
    const spreadsheet = obterPlanilhaOuAtiva_();
    const props = PropertiesService.getScriptProperties();
    if (!props.getProperty(ORION.propertySpreadsheetId)) {
      props.setProperty(ORION.propertySpreadsheetId, spreadsheet.getId());
    }

    // 1) PRE-FLIGHT: nenhuma escrita acontece antes desta inspeção terminar.
    const plano = inspecionarMigracaoV18_(spreadsheet);
    const assinaturaAntes = capturarAssinaturaDadosCriticos_(spreadsheet);

    // Se já estiver 100% em 1.8.0, apenas valida e encerra. Idempotência real.
    if (!plano.precisaMigrar) {
      validarEstrutura_(spreadsheet);
      validarIdsCriticos_(spreadsheet);
      Logger.log('Estrutura ' + ORION.schemaVersion + ' já estava aplicada. Nenhuma alteração realizada.');
      return {ok:true, ja_atualizado:true, schema_version:ORION.schemaVersion};
    }

    // 2) BACKUP COMPLETO: a migração só começa se o backup terminar com sucesso.
    backup = criarBackupCompletoOrion_(spreadsheet, 'PRE_V18');
    props.setProperty(ORION.propertyLastBackupId, backup.id);
    props.setProperty(ORION.propertyLastBackupUrl, backup.url);

    // 3) MIGRAÇÕES NÃO DESTRUTIVAS: somente inserção de colunas e criação de abas.
    migrarDividas_(spreadsheet, plano.dividas);
    migrarContasV17_(spreadsheet, plano.contas);
    migrarTransacoesV18_(spreadsheet, plano.transacoes);

    garantirAba_(spreadsheet,'EVENTOS',ORION.sheets.EVENTOS);
    garantirAba_(spreadsheet,'EVENTO_RESERVAS',ORION.sheets.EVENTO_RESERVAS);
    garantirAba_(spreadsheet,'PAGAMENTOS_DIVIDAS',ORION.sheets.PAGAMENTOS_DIVIDAS);
    garantirAba_(spreadsheet,'RADAR_INVESTIMENTOS',ORION.sheets.RADAR_INVESTIMENTOS);
    garantirAba_(spreadsheet,'METAS_FINANCEIRAS',ORION.sheets.METAS_FINANCEIRAS);
    garantirAba_(spreadsheet,'EVENTOS_XP',ORION.sheets.EVENTOS_XP);
    garantirAba_(spreadsheet,'TRANSFERENCIAS',ORION.sheets.TRANSFERENCIAS);
    garantirAba_(spreadsheet,'DESPESAS_RECORRENTES',ORION.sheets.DESPESAS_RECORRENTES);

    Object.keys(ORION.sheets).forEach(function(nome) {
      if (['DIVIDAS','PAGAMENTOS_DIVIDAS','RADAR_INVESTIMENTOS','METAS_FINANCEIRAS','EVENTOS_XP','TRANSFERENCIAS','DESPESAS_RECORRENTES','EVENTOS','EVENTO_RESERVAS'].indexOf(nome) >= 0) return;
      garantirAba_(spreadsheet,nome,ORION.sheets[nome]);
    });

    SpreadsheetApp.flush();

    // 4) VALIDAÇÃO COMPLETA ANTES de declarar a base como 1.8.0.
    validarEstrutura_(spreadsheet);
    validarIdsCriticos_(spreadsheet);
    const assinaturaDepois = capturarAssinaturaDadosCriticos_(spreadsheet);
    validarAssinaturaDadosCriticos_(assinaturaAntes, assinaturaDepois);

    // 5) META é atualizada SOMENTE após toda validação passar.
    atualizarMeta_(spreadsheet,'versao_estrutura',ORION.schemaVersion);
    atualizarMeta_(spreadsheet,'banco_atualizado_em',agoraIso_());
    atualizarMeta_(spreadsheet,'ultimo_backup_migracao_id',backup.id);
    atualizarMeta_(spreadsheet,'ultimo_backup_migracao_url',backup.url);
    SpreadsheetApp.flush();
    validarEstrutura_(spreadsheet);

    Logger.log('Estrutura atualizada para ' + ORION.schemaVersion + ' com sucesso.');
    Logger.log('Backup pré-migração: ' + backup.url);
    return {ok:true, schema_version:ORION.schemaVersion, backup_url:backup.url};
  } catch (erro) {
    const complemento = backup ? ' Backup preservado em: ' + backup.url : ' Nenhuma migração foi iniciada porque o backup não foi concluído.';
    Logger.log('MIGRAÇÃO INTERROMPIDA: ' + (erro.message || erro) + complemento);
    throw new Error('Migração interrompida por segurança. ' + (erro.message || String(erro)) + complemento);
  } finally {
    lock.releaseLock();
  }
}

function configurarTokenBrapi() {
  const ui = SpreadsheetApp.getUi();
  const resposta = ui.prompt(
    'Orion — chave brapi',
    'Cole a chave da brapi. Ela ficará nas propriedades privadas deste Apps Script.',
    ui.ButtonSet.OK_CANCEL
  );
  if (resposta.getSelectedButton() !== ui.Button.OK) return;
  const token = String(resposta.getResponseText() || '').trim();
  if (!token) throw new Error('Chave vazia.');
  PropertiesService.getScriptProperties().setProperty(ORION.propertyBrapiToken, token);
  ui.alert('Chave da brapi salva.');
}

function removerTokenBrapi() {
  PropertiesService.getScriptProperties().deleteProperty(ORION.propertyBrapiToken);
  SpreadsheetApp.getUi().alert('Chave da brapi removida.');
}

function doGet(e) {
  try {
    const params = (e && e.parameter) || {};
    const acao = String(params.acao || 'health').toLowerCase();

    if (acao === 'health') {
      return respostaJson_({
        ok:true, servico:'Orion Finance 2.0', api_version:'0.11.1',
        schema_version:ORION.schemaVersion, agora:agoraIso_()
      });
    }

    autenticar_(params.token);
    if (acao === 'versao') return respostaJson_(obterVersao_());
    if (acao === 'baixar') return respostaJson_(baixarTudo_());
    if (acao === 'exportar_xlsx') return respostaJson_(obterExportacaoXlsx_());
    if (acao === 'investimento') return respostaJson_(consultarInvestimento_(normalizarTicker_(params.ticker)));
    return respostaErro_('Ação GET desconhecida.',400);
  } catch (erro) {
    return respostaErro_(erro.message || String(erro),500);
  }
}

function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) return respostaErro_('Corpo da requisição ausente.',400);
    const payload = JSON.parse(e.postData.contents);
    autenticar_(payload.token);
    const acao = String(payload.acao || '').toLowerCase();
    if (acao === 'sincronizar') return respostaJson_(sincronizar_(payload));
    if (acao === 'zerar_dados') return respostaJson_(zerarDadosOrion_());
    return respostaErro_('Ação POST desconhecida.',400);
  } catch (erro) {
    return respostaErro_(erro.message || String(erro),500);
  }
}

function consultarInvestimento_(ticker) {
  if (!ticker) throw new Error('Ticker não informado.');
  const brapiToken = PropertiesService.getScriptProperties().getProperty(ORION.propertyBrapiToken) || '';
  const sandbox = ['PETR4','VALE3','MGLU3','ITUB4'].indexOf(ticker) >= 0;

  if (!brapiToken && !sandbox) {
    return {
      ok:false, precisa_brapi:true, ticker:ticker,
      erro:'Para consultar este ativo, configure uma chave da brapi no Apps Script.'
    };
  }

  const base = 'https://brapi.dev/api/quote/' + encodeURIComponent(ticker);
  const params = [
    'dividends=true',
    'modules=summaryProfile,balanceSheet,incomeStatement,financialData,defaultKeyStatistics'
  ];
  if (brapiToken) params.push('token=' + encodeURIComponent(brapiToken));

  let payload;
  let detalhado = true;
  try {
    payload = buscarJsonExterno_(base + '?' + params.join('&'));
  } catch (erro) {
    detalhado = false;
    const fallback = brapiToken ? base + '?token=' + encodeURIComponent(brapiToken) : base;
    payload = buscarJsonExterno_(fallback);
  }

  const item = payload && payload.results && payload.results[0];
  if (!item) throw new Error('Ativo não encontrado na fonte de mercado.');

  return {
    ok:true, ticker:ticker, detalhado:detalhado,
    consultado_em:agoraIso_(), fonte:'brapi.dev', dados:item
  };
}

function buscarJsonExterno_(url) {
  const response = UrlFetchApp.fetch(url,{
    method:'get', muteHttpExceptions:true, followRedirects:true
  });
  const code = response.getResponseCode();
  const text = response.getContentText();
  if (code < 200 || code >= 300) {
    let message = 'Fonte de mercado retornou HTTP ' + code + '.';
    try {
      const parsed = JSON.parse(text);
      message = parsed.message || parsed.error || message;
    } catch (_) {}
    throw new Error(message);
  }
  return JSON.parse(text);
}

function normalizarTicker_(valor) {
  return String(valor || '').toUpperCase().replace(/[^A-Z0-9]/g,'').slice(0,12);
}


function zerarDadosOrion_() {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    const spreadsheet = obterPlanilha_();
    validarEstrutura_(spreadsheet);

    const preservar = ['META', 'INSTITUICOES', 'CATEGORIAS', 'SUBCATEGORIAS', 'FRASES'];
    Object.keys(ORION.sheets).forEach(function(nome) {
      if (preservar.indexOf(nome) >= 0) return;
      const sheet = spreadsheet.getSheetByName(nome);
      if (!sheet) return;
      const lastRow = sheet.getLastRow();
      if (lastRow >= 2) {
        sheet.getRange(2, 1, lastRow - 1, Math.max(sheet.getLastColumn(), ORION.sheets[nome].length)).clearContent();
      }
    });

    atualizarMeta_(spreadsheet, 'banco_atualizado_em', agoraIso_());
    atualizarMeta_(spreadsheet, 'ultima_sincronizacao_em', '');
    atualizarMeta_(spreadsheet, 'ultimo_dispositivo_sync', '');

    return {
      ok:true,
      zerado:true,
      schema_version:ORION.schemaVersion,
      banco_atualizado_em:obterMeta_(spreadsheet,'banco_atualizado_em') || '',
      agora:agoraIso_()
    };
  } finally {
    lock.releaseLock();
  }
}

function sincronizar_(payload) {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    const spreadsheet = obterPlanilha_();
    validarEstrutura_(spreadsheet);

    const remoto = obterMeta_(spreadsheet,'banco_atualizado_em') || '';
    const cliente = String(payload.base_atualizada_em || '');
    const alteracoes = payload.alteracoes || {};
    const dispositivo = limparTexto_(payload.dispositivo || 'desconhecido',80);
    const total = contarAlteracoes_(alteracoes);

    if (total > ORION.maxPayloadRows) throw new Error('Quantidade de alterações acima do limite.');
    if (total > 0 && cliente && remoto && cliente !== remoto) {
      return {
        ok:false, conflito:true,
        mensagem:'A base online mudou desde a última sincronização.',
        banco_atualizado_em:remoto,
        dados:lerTodasTabelas_(spreadsheet)
      };
    }

    if (total > 0) {
      aplicarAlteracoes_(spreadsheet,alteracoes);
      atualizarMeta_(spreadsheet,'banco_atualizado_em',agoraIso_());
    }

    atualizarMeta_(spreadsheet,'ultima_sincronizacao_em',agoraIso_());
    atualizarMeta_(spreadsheet,'ultimo_dispositivo_sync',dispositivo);

    return {
      ok:true, conflito:false, schema_version:ORION.schemaVersion,
      banco_atualizado_em:obterMeta_(spreadsheet,'banco_atualizado_em') || '',
      sincronizado_em:agoraIso_(), dados:lerTodasTabelas_(spreadsheet)
    };
  } finally {
    lock.releaseLock();
  }
}

function lerTodasTabelas_(spreadsheet) {
  const resultado = {};
  Object.keys(ORION.sheets).forEach(function(nome) {
    if (nome === 'META') return;
    resultado[nome] = lerTabela_(spreadsheet.getSheetByName(nome));
  });
  return resultado;
}

function lerTabela_(sheet) {
  if (!sheet) return [];
  const lastRow = sheet.getLastRow();
  const lastColumn = sheet.getLastColumn();
  if (lastRow < 2 || lastColumn < 1) return [];

  const values = sheet.getRange(1,1,lastRow,lastColumn).getValues();
  const headers = values[0].map(String);
  const resultado = [];

  values.slice(1).forEach(function(row) {
    if (row.every(function(v) { return v === '' || v === null; })) return;
    const item = {};
    headers.forEach(function(header,index) { item[header] = serializarValor_(row[index]); });
    resultado.push(item);
  });
  return resultado;
}

function aplicarAlteracoes_(spreadsheet,alteracoes) {
  Object.keys(alteracoes).forEach(function(nomeAba) {
    if (!Object.prototype.hasOwnProperty.call(ORION.idFields,nomeAba)) throw new Error('Tabela não permitida: ' + nomeAba);
    const registros = alteracoes[nomeAba];
    if (!Array.isArray(registros)) throw new Error('Alterações inválidas em ' + nomeAba + '.');
    if (!registros.length) return;
    upsertRegistros_(spreadsheet.getSheetByName(nomeAba),nomeAba,registros);
  });
}

function upsertRegistros_(sheet,nomeAba,registros) {
  const headers = ORION.sheets[nomeAba];
  const idField = ORION.idFields[nomeAba];
  const idColumn = headers.indexOf(idField) + 1;
  const lastRow = Math.max(sheet.getLastRow(),1);
  const mapa = {};

  validarCabecalho_(nomeAba,sheet.getRange(1,1,1,headers.length).getValues()[0].map(String),headers);

  if (lastRow >= 2) {
    sheet.getRange(2,idColumn,lastRow - 1,1).getValues().forEach(function(row,index) {
      const id = String(row[0] || '').trim();
      if (id) mapa[id] = index + 2;
    });
  }

  const novas = [];
  const updates = [];

  registros.forEach(function(registro) {
    const id = limparTexto_(registro[idField],160);
    if (!id) throw new Error(nomeAba + ': registro sem ' + idField + '.');
    const row = headers.map(function(h) { return normalizarValorParaPlanilha_(registro[h]); });
    if (mapa[id]) updates.push({linha:mapa[id],valores:row});
    else novas.push(row);
  });

  updates.forEach(function(item) {
    sheet.getRange(item.linha,1,1,headers.length).setValues([item.valores]);
  });

  if (novas.length) {
    sheet.getRange(sheet.getLastRow() + 1,1,novas.length,headers.length).setValues(novas);
  }
}

function cabecalhoCorresponde_(sheet,headers) {
  if (!sheet) return false;
  const atual = sheet.getRange(1,1,1,headers.length).getValues()[0].map(function(v){ return String(v || '').trim(); });
  return headers.every(function(h,i){ return atual[i] === String(h).trim(); });
}

function inspecionarMigracaoV18_(spreadsheet) {
  const legacy = {
    DIVIDAS:['divida_id','descricao','credor','valor_original','valor_atual','total_parcelas','parcela_atual','vencimento','status','observacoes','criado_em','atualizado_em'],
    CONTAS:['conta_id','instituicao_id','nome','tipo','saldo_inicial','saldo_atual','status','criado_em','atualizado_em'],
    TRANSACOES:['transacao_id','data','descricao','valor','tipo','categoria_id','subcategoria_id','conta_id','forma_pagamento','cartao_id','observacao','criado_em','atualizado_em']
  };
  function modo(nome) {
    const sheet = spreadsheet.getSheetByName(nome);
    if (!sheet) return 'ausente';
    if (cabecalhoCorresponde_(sheet,ORION.sheets[nome])) return 'atual';
    if (legacy[nome] && cabecalhoCorresponde_(sheet,legacy[nome])) return 'legado';
    throw new Error('Pré-validação falhou em ' + nome + ': o cabeçalho não corresponde nem ao schema atual nem ao schema legado reconhecido. Nenhum dado foi alterado.');
  }

  // Abas que não precisam de transformação devem, se existirem, já ter cabeçalho conhecido.
  Object.keys(ORION.sheets).forEach(function(nome) {
    if (['DIVIDAS','CONTAS','TRANSACOES','EVENTOS','EVENTO_RESERVAS'].indexOf(nome) >= 0) return;
    const sheet = spreadsheet.getSheetByName(nome);
    if (!sheet) return;
    const esperado = ORION.sheets[nome];
    const atual = sheet.getRange(1,1,1,esperado.length).getValues()[0].map(String);
    validarCabecalho_(nome,atual,esperado);
  });

  const plano = {dividas:modo('DIVIDAS'), contas:modo('CONTAS'), transacoes:modo('TRANSACOES')};
  const metaAtual = obterMeta_(spreadsheet,'versao_estrutura');
  const eventosOk = cabecalhoCorresponde_(spreadsheet.getSheetByName('EVENTOS'),ORION.sheets.EVENTOS);
  const reservasOk = cabecalhoCorresponde_(spreadsheet.getSheetByName('EVENTO_RESERVAS'),ORION.sheets.EVENTO_RESERVAS);
  plano.precisaMigrar = plano.dividas !== 'atual' || plano.contas !== 'atual' || plano.transacoes !== 'atual' || !eventosOk || !reservasOk || metaAtual !== ORION.schemaVersion;
  return plano;
}

function criarBackupCompletoOrion_(spreadsheet,rotulo) {
  const stamp = Utilities.formatDate(new Date(), spreadsheet.getSpreadsheetTimeZone() || Session.getScriptTimeZone(), 'yyyyMMdd_HHmmss');
  const nome = 'Orion Finance BACKUP ' + String(rotulo || 'MANUAL') + ' ' + stamp;
  const backup = SpreadsheetApp.create(nome);
  backup.setSpreadsheetTimeZone(spreadsheet.getSpreadsheetTimeZone());
  const manifest = backup.getSheets()[0];
  manifest.setName('BACKUP_MANIFESTO');

    ['origem_nome',spreadsheet.getName()],
    ['criado_em',agoraIso_()],
    ['schema_origem',obterMeta_(spreadsheet,'versao_estrutura') || 'não informado'],
    ['rotulo',String(rotulo || '')]
  ]);

  spreadsheet.getSheets().forEach(function(source) {
    const nomeAba = source.getName();
    if (nomeAba === 'BACKUP_MANIFESTO') return;
    const target = backup.insertSheet(nomeAba.slice(0,99));
    const range = source.getDataRange();
    const values = range.getValues();
    if (values.length && values[0].length) target.getRange(1,1,values.length,values[0].length).setValues(values);
    target.setFrozenRows(source.getFrozenRows());
  });
  SpreadsheetApp.flush();
  return {id:backup.getId(), url:backup.getUrl(), nome:nome, criado_em:agoraIso_()};
}

function restaurarUltimoBackupOrion() {
  const ui = SpreadsheetApp.getUi();
  const resposta = ui.prompt(
    'ORION — RESTAURAÇÃO DE EMERGÊNCIA',
    'Esta função substituirá os dados atuais pelos valores do último backup automático. Antes disso, outro backup de segurança será criado. Digite RESTAURAR para continuar.',
    ui.ButtonSet.OK_CANCEL
  );
  if (resposta.getSelectedButton() !== ui.Button.OK || String(resposta.getResponseText() || '').trim().toUpperCase() !== 'RESTAURAR') {
    ui.alert('Restauração cancelada. Nenhum dado foi alterado.');
    return;
  }

  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    const spreadsheet = obterPlanilhaOuAtiva_();
    const props = PropertiesService.getScriptProperties();
    const backupId = props.getProperty(ORION.propertyLastBackupId);
    if (!backupId) throw new Error('Nenhum backup automático foi localizado.');
    const backup = SpreadsheetApp.openById(backupId);

    // Cria uma cópia do estado atual antes da restauração.
    criarBackupCompletoOrion_(spreadsheet,'PRE_RESTORE');

    backup.getSheets().forEach(function(source) {
      const nome = source.getName();
      if (nome === 'BACKUP_MANIFESTO') return;
      let target = spreadsheet.getSheetByName(nome);
      if (!target) target = spreadsheet.insertSheet(nome);
      const values = source.getDataRange().getValues();
      const rows = Math.max(values.length,1);
      const cols = Math.max(values[0] ? values[0].length : 1,1);
      if (target.getMaxColumns() < cols) target.insertColumnsAfter(target.getMaxColumns(), cols - target.getMaxColumns());
      if (target.getMaxRows() < rows) target.insertRowsAfter(target.getMaxRows(), rows - target.getMaxRows());
      target.clearContents();
      if (values.length && values[0].length) target.getRange(1,1,values.length,values[0].length).setValues(values);
      if (target.getMaxColumns() > cols) target.deleteColumns(cols + 1, target.getMaxColumns() - cols);
    });
    SpreadsheetApp.flush();
    ui.alert('Backup restaurado. Revise a planilha antes de sincronizar o aplicativo.');
  } finally {
    lock.releaseLock();
  }
}

function validarIdsCriticos_(spreadsheet) {
  ['CONTAS','TRANSACOES','DIVIDAS'].forEach(function(nome) {
    const sheet = spreadsheet.getSheetByName(nome);
    if (!sheet || sheet.getLastRow() < 2) return;
    const ids = sheet.getRange(2,1,sheet.getLastRow()-1,1).getValues().map(function(r){return String(r[0]||'').trim();}).filter(Boolean);
    const vistos = {};
    ids.forEach(function(id){ if (vistos[id]) throw new Error(nome + ': ID duplicado detectado (' + id + '). Migração interrompida.'); vistos[id]=true; });
  });
}

function digestIds_(ids) {
  const bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, ids.slice().sort().join('|'), Utilities.Charset.UTF_8);
  return bytes.map(function(b){ const v=(b+256)%256; return ('0'+v.toString(16)).slice(-2); }).join('');
}

function numeroSeguro_(v) { const n=Number(v); return Number.isFinite(n) ? n : 0; }

function assinaturaTabela_(sheet,nome,camposSoma) {
  if (!sheet || sheet.getLastRow() < 2) return {nome:nome,linhas:0,digest:digestIds_([]),somas:{}};
  const cols = Math.max(sheet.getLastColumn(),1);
  const headers = sheet.getRange(1,1,1,cols).getValues()[0].map(function(v){return String(v||'').trim();});
  const rows = sheet.getRange(2,1,sheet.getLastRow()-1,cols).getValues();
  const ids = rows.map(function(r){return String(r[0]||'').trim();}).filter(Boolean);
  const somas = {};
  (camposSoma || []).forEach(function(campo){
    const idx=headers.indexOf(campo);
    somas[campo]=idx<0?0:rows.reduce(function(s,r){return s+numeroSeguro_(r[idx]);},0);
  });
  return {nome:nome,linhas:ids.length,digest:digestIds_(ids),somas:somas};
}

function capturarAssinaturaDadosCriticos_(spreadsheet) {
  return {

    TRANSACOES:assinaturaTabela_(spreadsheet.getSheetByName('TRANSACOES'),'TRANSACOES',['valor']),
    DIVIDAS:assinaturaTabela_(spreadsheet.getSheetByName('DIVIDAS'),'DIVIDAS',['valor_original','valor_atual'])
  };
}

function validarAssinaturaDadosCriticos_(antes,depois) {
  ['CONTAS','TRANSACOES','DIVIDAS'].forEach(function(nome){
    const a=antes[nome], d=depois[nome];
    if (a.linhas !== d.linhas) throw new Error(nome + ': quantidade de registros mudou durante a migração (' + a.linhas + ' → ' + d.linhas + ').');
    if (a.digest !== d.digest) throw new Error(nome + ': conjunto de IDs mudou durante a migração.');
    Object.keys(a.somas).forEach(function(campo){
      if (Math.abs(numeroSeguro_(a.somas[campo]) - numeroSeguro_(d.somas[campo])) > 0.001) {
        throw new Error(nome + ': soma de ' + campo + ' mudou durante a migração.');
      }
    });
  });
}

function migrarDividas_(spreadsheet,modo) {
  let sheet = spreadsheet.getSheetByName('DIVIDAS');
  if (!sheet) { garantirAba_(spreadsheet,'DIVIDAS',ORION.sheets.DIVIDAS); return; }
  if (modo === 'atual') return;
  if (modo !== 'legado') throw new Error('DIVIDAS: migração não autorizada para schema desconhecido.');

  // Legado 12 colunas -> atual 16 colunas. Apenas INSERE colunas; nenhum clear().
  sheet.insertColumnAfter(3); // tipo
  sheet.getRange(1,4).setValue('tipo');
  sheet.insertColumnsAfter(6,2); // forma_pagamento + valor_pagamento_planejado
  sheet.getRange(1,7,1,2).setValues([['forma_pagamento','valor_pagamento_planejado']]);
  sheet.insertColumnAfter(11); // proximo_vencimento
  sheet.getRange(1,12).setValue('proximo_vencimento');

  const lastRow=sheet.getLastRow();
  if (lastRow>=2) {
    sheet.getRange(2,4,lastRow-1,1).setValues(Array.from({length:lastRow-1},function(){return ['informal'];}));
    sheet.getRange(2,7,lastRow-1,1).setValues(Array.from({length:lastRow-1},function(){return ['livre'];}));

  formatarCabecalho_(sheet,ORION.sheets.DIVIDAS.length);
  validarCabecalho_('DIVIDAS',sheet.getRange(1,1,1,ORION.sheets.DIVIDAS.length).getValues()[0].map(String),ORION.sheets.DIVIDAS);
}

function garantirAba_(spreadsheet,nome,headers) {
  let sheet = spreadsheet.getSheetByName(nome);
  if (!sheet) sheet = spreadsheet.insertSheet(nome);

  if (sheet.getLastRow() === 0) {
    sheet.getRange(1,1,1,headers.length).setValues([headers]);
  } else {
    const atual = sheet.getRange(1,1,1,headers.length).getValues()[0].map(String);
    if (atual.every(function(v) { return !v; })) sheet.getRange(1,1,1,headers.length).setValues([headers]);
  }
  formatarCabecalho_(sheet,headers.length);
  return sheet;
}

function formatarCabecalho_(sheet,columns) {
  sheet.setFrozenRows(1);
  sheet.getRange(1,1,1,columns).setFontWeight('bold').setFontColor('#ffffff').setBackground('#111827');
}

function validarEstrutura_(spreadsheet) {
  Object.keys(ORION.sheets).forEach(function(nomeAba) {
    const sheet = spreadsheet.getSheetByName(nomeAba);
    if (!sheet) throw new Error('Aba obrigatória ausente: ' + nomeAba);
    const esperado = ORION.sheets[nomeAba];
    const atual = sheet.getRange(1,1,1,esperado.length).getValues()[0].map(String);
    validarCabecalho_(nomeAba,atual,esperado);
  });
  return true;
}

function validarCabecalho_(nomeAba,atual,esperado) {
  for (let i = 0; i < esperado.length; i++) {
    if (String(atual[i] || '').trim() !== String(esperado[i]).trim()) {
      throw new Error('Cabeçalho incompatível em ' + nomeAba + ', coluna ' + (i + 1) + '. Esperado "' + esperado[i] + '", encontrado "' + (atual[i] || '') + '".');
    }
  }
}

function atualizarMeta_(spreadsheet,chave,valor) {
  const sheet = spreadsheet.getSheetByName('META');
  const lastRow = Math.max(sheet.getLastRow(),1);
  if (lastRow >= 2) {
    const keys = sheet.getRange(2,1,lastRow - 1,1).getValues();
    for (let i = 0; i < keys.length; i++) {
      if (String(keys[i][0]).trim() === chave) {
        sheet.getRange(i + 2,2).setValue(valor);
        return;
      }
    }
  }
  sheet.appendRow([chave,valor,'']);
}

function obterMeta_(spreadsheet,chave) {
  const sheet = spreadsheet.getSheetByName('META');
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return '';
  const values = sheet.getRange(2,1,lastRow - 1,2).getValues();
  for (let i = 0; i < values.length; i++) {
    if (String(values[i][0]).trim() === chave) return serializarValor_(values[i][1]);
  }
  return '';
}

function contarAlteracoes_(alteracoes) {
  if (!alteracoes || typeof alteracoes !== 'object' || Array.isArray(alteracoes)) return 0;
  return Object.keys(alteracoes).reduce(function(total,chave) {
    return total + (Array.isArray(alteracoes[chave]) ? alteracoes[chave].length : 0);
  },0);
}

function normalizarValorParaPlanilha_(value) {
  if (value === undefined || value === null) return '';
  if (value instanceof Date) return value;
  if (typeof value === 'number' || typeof value === 'boolean') return value;
  if (typeof value === 'string') return value.length > 5000 ? value.slice(0,5000) : value;
  return JSON.stringify(value);
}

function serializarValor_(value) {
  if (value instanceof Date) {
    return Utilities.formatDate(value,Session.getScriptTimeZone() || 'America/Sao_Paulo',"yyyy-MM-dd'T'HH:mm:ssXXX");
  }
  return value;
}

function limparTexto_(value,limite) {
  return String(value === undefined || value === null ? '' : value).trim().slice(0,limite || 500);
}
